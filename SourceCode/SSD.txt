Function ObterDispositivos()
{
	DriveList = DriveInfo:GetDrives()
	ForEach (Drive, DriveList)
	{
		If (Drive:DriveType = DriveType:Fixed)
		{
			If (Drive:VolumeLabel = "Luis Costa Storage 3") { Dispositivos.DiscoBeatriz = Drive.Name }
			If (Drive:VolumeLabel = "Sistema") { Dispositivos.DiscoLuis = Drive.Name }
			If (Drive:VolumeLabel = "Luis Costa Storage 2") { Dispositivos.DiscoSoftware = Drive.Name }
			If (Drive:VolumeLabel = "Documentos") { Dispositivos.DiscoMovel = Drive.Name }
		}
	}
}
Function ReportarDispositivos()
{
	Host:Write("                      Disco Beatriz > ")
	Host:WriteLine(Dispositivos.DiscoBeatriz)
	Host:Write("                         Disco Luís > ")
	Host:WriteLine(Dispositivos.DiscoLuis)
	Host:Write("                     Disco Software > ")
	Host:WriteLine(Dispositivos.DiscoSoftware)
	Host:Write("                        Disco Móvel > ")
	Host:WriteLine(Dispositivos.DiscoMovel)
}
Function ObterPacotes()
{
	If (Dispositivos.DiscoMovel = "") { Return False }

	If (Dispositivos.DiscoBeatriz != "") { Pacotes.FMSBeatriz = "Sim" }
	If (Dispositivos.DiscoLuis <> "") { Pacotes.FMSLuis = "Sim" }
	If (Dispositivos.DiscoSoftware != "")
	{
		Pacotes.VirtualMachine = "Sim"
		Pacotes.Software= "Sim"
	}
}

	Host:Write("                    Virtual Machine > ")

	Host:Write("                           Software > ")
	Host:WriteLine(Pacotes.Software)
}
Function PrepararPacoteFMSLuis()
{
	Folder = CreateAssemblyObject("C:\WINDOWS\Microsoft.NET\Framework\v2.0.50727\mscorlib.dll", "System.IO.DirectoryInfo", "D:\TesteOrigem")
	CalcularTotalBytes(Folder)
}
Function CalcularTotalBytes(Folder)
{
	Directorias = Folder:GetDirectories()
	ForEach (Directoria, Directorias) { CalcularTotalBytes(Directoria) }
	Ficheiros = Folder:GetFiles()
	ForEach (Ficheiro, Ficheiros) { Add(TotalBytes, Ficheiro.Length) }
}

Function ExecutarPacoteFMSLuis()
{
	Folder = CreateAssemblyObject("C:\WINDOWS\Microsoft.NET\Framework\v2.0.50727\mscorlib.dll", "System.IO.DirectoryInfo", "D:\TesteOrigem")
	ProcessarDirectoria(Folder, "D:\TesteDestino")
}
Function ProcessarDirectoria(Origem, Destino)
{
	OrigemName = Concat(Origem:FullName, "\")
	DestinoName = Concat(Destino, "\")
	
	Ficheiros = Origem:GetFiles()
	ForEach (Ficheiro, Ficheiros) 
	{
		Add(TotalBytesCopiados, Ficheiro:Length)
		FileName = Ficheiro:Name
		File:Move(Concat(OrigemName, FileName), Concat(DestinoName, FileName))
		TotalBytesCopiadosAnterior = Host:ProgressBar("A executar: ", TotalBytesCopiados, TotalBytes, TotalBytesCopiadosAnterior)
	}
	
	Directorias = Origem:GetDirectories()
	ForEach (Directoria, Directorias)
	{
		Aux = Concat(DestinoName, Directoria.Name)
		If (Directory:Exists(Aux) = False) { Directory:CreateDirectory(Aux) }
		ProcessarDirectoria(Directoria, Aux)
		Directory:Delete(Directoria:FullName)
	}
}
